-- Trigger de sincronizacion
CREATE OR REPLACE FUNCTION handle_new_user()
RETURNS trigger AS $$
BEGIN
  INSERT INTO usuarios (id_usuario, nombre, correo, rol, fecha_registro)
  VALUES (
    NEW.id,
    COALESCE(NEW.raw_user_meta_data->>'nombre', 'Sin nombre'),
    NEW.email,
    'ciudadano',
    NOW()
  );
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER on_auth_user_created
AFTER INSERT ON auth.users
FOR EACH ROW EXECUTE FUNCTION handle_new_user();

-- Activar Row Level Security en tablas sensibles
ALTER TABLE usuarios ENABLE ROW LEVEL SECURITY;
ALTER TABLE reportes ENABLE ROW LEVEL SECURITY;
ALTER TABLE adjuntos ENABLE ROW LEVEL SECURITY;
ALTER TABLE comentarios ENABLE ROW LEVEL SECURITY;
ALTER TABLE asignaciones ENABLE ROW LEVEL SECURITY;
ALTER TABLE seguimiento ENABLE ROW LEVEL SECURITY;

-- POLÍTICAS RLS

-- 1. USUARIOS
CREATE POLICY "Ciudadano ve su perfil"
ON usuarios
FOR SELECT
USING (id_usuario = auth.uid());

CREATE POLICY "Ciudadano edita su perfil"
ON usuarios
FOR UPDATE
USING (id_usuario = auth.uid());

CREATE POLICY "Admin acceso total usuarios"
ON usuarios
FOR ALL
USING (
  EXISTS (
    SELECT 1 FROM usuarios u
    WHERE u.id_usuario = auth.uid()
    AND u.rol = 'admin'
  )
);

-- 2. REPORTES
CREATE POLICY "Ciudadano crea reporte"
ON reportes
FOR INSERT
WITH CHECK (usuario_id = auth.uid());

CREATE POLICY "Ciudadano ve sus reportes"
ON reportes
FOR SELECT
USING (usuario_id = auth.uid());

CREATE POLICY "Funcionario ve reportes de su area"
ON reportes
FOR SELECT
USING (
    categoria_id IN (
        SELECT id_categoria FROM categorias 
        WHERE area_id = (
            SELECT area_id FROM usuarios WHERE id_usuario = auth.uid()
        )
    )
);

CREATE POLICY "Admin acceso total reportes"
ON reportes
FOR ALL
USING (
  EXISTS (
    SELECT 1 FROM usuarios u
    WHERE u.id_usuario = auth.uid()
    AND u.rol = 'admin'
  )
);

INSERT INTO storage.buckets (id, name, public) VALUES 
('fotos', 'fotos', true),
('videos', 'videos', true),
('documentos', 'documentos', true)
ON CONFLICT (id) DO NOTHING;

-- Política: cualquier usuario autenticado puede subir archivos a esos buckets
CREATE POLICY "Usuarios autenticados pueden subir evidencia"
ON storage.objects
FOR INSERT TO authenticated
WITH CHECK (bucket_id IN ('fotos', 'videos', 'documentos') AND owner = auth.uid());

-- Política: el dueño puede ver sus propios archivos
CREATE POLICY "Dueño ve sus archivos"
ON storage.objects
FOR SELECT TO authenticated
USING (owner = auth.uid());

-- Política: admin acceso total a storage
CREATE POLICY "Admin acceso total storage"
ON storage.objects
FOR ALL TO authenticated
USING (
  EXISTS (
    SELECT 1 FROM usuarios u
    WHERE u.id_usuario = auth.uid()
    AND u.rol = 'admin'
  )
);
