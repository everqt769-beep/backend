CREATE EXTENSION IF NOT EXISTS "uuid-ossp";


-- 1. ESTADOS (Catalogo)
CREATE TABLE estados (
    id_estado UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    entidad TEXT NOT NULL, -- 'reporte', 'asignacion', 'usuario', 'adjunto'
    codigo TEXT UNIQUE NOT NULL, -- 'pendiente', 'en_proceso', 'resuelto', 'activo', 'inapropiado'
    nombre TEXT NOT NULL,
    color VARCHAR(7), -- Para el frontend (#FFFFFF)
    descripcion TEXT
);

ALTER TABLE estados ADD CONSTRAINT check_color_valid 
    CHECK (color IS NULL OR color ~ '^#[0-9A-Fa-f]{6}$');

-- 2. AREAS (Normalizacion)
CREATE TABLE areas (
    id_area UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    nombre TEXT UNIQUE NOT NULL,
    descripcion TEXT,
    telefono VARCHAR(20),
    email TEXT
);

-- 3. USUARIOS (Sincronizado con Supabase Auth)
CREATE TABLE usuarios (
    id_usuario UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    nombre TEXT NOT NULL,
    correo TEXT UNIQUE NOT NULL,
    rol TEXT CHECK (rol IN ('ciudadano', 'funcionario', 'admin')) DEFAULT 'ciudadano',
    area_id UUID REFERENCES areas(id_area),
    telefono VARCHAR(20),
    estado_id UUID REFERENCES estados(id_estado),
    fecha_registro TIMESTAMPTZ DEFAULT NOW()
);

-- 4. CATEGORIAS
CREATE TABLE categorias (
    id_categoria UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    nombre TEXT NOT NULL,
    descripcion TEXT,
    prioridad_base INT CHECK (prioridad_base BETWEEN 1 AND 3),
    area_id UUID REFERENCES areas(id_area)
);

-- 5. REPORTES
CREATE TABLE reportes (
    id_reporte UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    usuario_id UUID REFERENCES usuarios(id_usuario) ON DELETE CASCADE,
    categoria_id UUID REFERENCES categorias(id_categoria),
    descripcion TEXT NOT NULL,
    latitud DECIMAL(10, 7),
    longitud DECIMAL(10, 7),
    direccion TEXT,
    estado_id UUID REFERENCES estados(id_estado),
    fecha_creacion TIMESTAMPTZ DEFAULT NOW()
);

-- 6. ADJUNTOS
CREATE TABLE adjuntos (
    id_adjunto UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    reporte_id UUID REFERENCES reportes(id_reporte) ON DELETE CASCADE,
    usuario_id UUID REFERENCES usuarios(id_usuario),
    tipo TEXT CHECK (tipo IN ('imagen', 'video', 'documento')),
    url TEXT NOT NULL, -- URL de Supabase Storage
    estado_moderacion_id UUID REFERENCES estados(id_estado),
    nombre_archivo TEXT,
    tamano_bytes BIGINT,
    descripcion TEXT,
    fecha_subida TIMESTAMPTZ DEFAULT NOW()
);

-- 7. ASIGNACIONES
CREATE TABLE asignaciones (
    id_asignacion UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    reporte_id UUID REFERENCES reportes(id_reporte) ON DELETE CASCADE,
    funcionario_id UUID REFERENCES usuarios(id_usuario),
    fecha_asignacion TIMESTAMPTZ DEFAULT NOW(),
    estado_id UUID REFERENCES estados(id_estado)
);

-- 8. COMENTARIOS
CREATE TABLE comentarios (
    id_comentario UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    reporte_id UUID REFERENCES reportes(id_reporte) ON DELETE CASCADE,
    usuario_id UUID REFERENCES usuarios(id_usuario),
    texto TEXT NOT NULL,
    tipo TEXT CHECK (tipo IN ('publico', 'interno', 'respuesta_oficial')),
    padre_id UUID REFERENCES comentarios(id_comentario) ON DELETE SET NULL, --  corregido
    fecha TIMESTAMPTZ DEFAULT NOW()
);

-- 9. SEGUIMIENTO
CREATE TABLE seguimiento (
    id_seguimiento UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    reporte_id UUID REFERENCES reportes(id_reporte) ON DELETE CASCADE,
    usuario_id UUID REFERENCES usuarios(id_usuario),
    tipo_evento TEXT,
    descripcion TEXT,
    estado_nuevo_id UUID REFERENCES estados(id_estado),
    fecha TIMESTAMPTZ DEFAULT NOW()
);
